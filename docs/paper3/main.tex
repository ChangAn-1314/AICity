\newcommand{\setpageframe}{%
  \AddToShipoutPictureFG{%
    \begin{tikzpicture}[remember picture,overlay]
      \ifshowpageframe
      \fill[headercharcoal]
        ($(current page.north west)+(0,-\pageheadercenterY+0.19cm)$)
        rectangle
        ($(current page.north east)+(0,-\pageheadercenterY-0.19cm)$);
      \node[
        anchor=west,
        text=white,
        font=\heiti\fontsize{10.5pt}{10.5pt}\selectfont
      ] at ($(current page.north west)+(3cm,-\pageheadercenterY)$) {智舆};
      \node[
        regular polygon,
        regular polygon sides=6,
        shape border rotate=30,
        shade,
        left color=pagehexcyandark,
        right color=pagehexcyanlight,
        draw=none,
        minimum size=\pagehexsize,
        inner sep=0pt
      ] (pagenumberbadge) at ($(current page.north east)+(-1.76cm,-\pageheadercenterY)$) {};
      \node[
        text=white,
        font=\bfseries\fontsize{12pt}{12pt}\selectfont
      ] at (pagenumberbadge.center) {\thepage};
      \node[
        anchor=south east,
        inner sep=0pt,
        opacity=0.14
      ] at ($(current page.south east)+(-0.62cm,0.28cm)$) {%
        \includegraphics[width=\pagewatermarkwidth]{logonew.png}%
      };
      \node[
        anchor=south east,
        inner sep=0pt
      ] at ($(current page.south east)+(-0.68cm,0.34cm)$) {%
        \includegraphics[width=\pagewatermarkwidth]{logonew.png}%
      };
      \fi
    \end{tikzpicture}%
  }%
}
